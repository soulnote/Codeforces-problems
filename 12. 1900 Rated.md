```java
                arr[i][1] = sc.nextInt();

                if (x[0] > lastA) lastA = x[0];
                if (x[1] < lastB) ans += lastB - x[1];
                else lastB = x[1];
            }
            System.out.println(ans);
        }
```

## **9. Non-academic Problem (Problem 2049I)**

**Concept:**  
Find connected components, answer = sum(size choose 2).

```java
import java.util.*;

public class NonAcademicProblem {
    static List<Integer>[] graph;
    static boolean[] visited;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            for (int i = 0; i < m; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                graph[u].add(v);
                graph[v].add(u);
            }
            visited = new boolean[n + 1];
            long ans = 0;
            for (int i = 1; i <= n; i++) {
                if (!visited[i]) {
                    int size = dfs(i);
                    ans += (long) size * (size - 1) / 2;
                }
            }
            System.out.println(ans);
        }
    }
    private static int dfs(int node) {
        visited[node] = true;
        int size = 1;
        for (int child : graph[node]) if (!visited[child]) size += dfs(child);
        return size;
    }
}
```

## **11. Shuffling Songs (Problem 2049K)**

**Concept:**  
Count permutations where adjacent differ. Use DP.

```java
import java.util.*;

public class ShufflingSongs {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String[] songs = new String[n];
            for (int i = 0; i < n; i++) songs[i] = sc.next();
            boolean[][] adj = new boolean[n][n];
            for (int i = 0; i < n; i++)
                for (int j = 0; j < n; j++)
                    if (i != j && songs[i].charAt(0) == songs[j].charAt(0) ||
                        songs[i].charAt(songs[i].length() - 1) == songs[j].charAt(songs[j].length() - 1))
                        adj[i][j] = true;
            long[][] dp = new long[n][1 << n];
            for (int i = 0; i < n; i++) dp[i][1 << i] = 1;
            for (int mask = 1; mask < (1 << n); mask++)
                for (int last = 0; last < n; last++)
                    if ((mask >> last & 1) == 1 && dp[last][mask] > 0)
                        for (int next = 0; next < n; next++)
                            if ((mask >> next & 1) == 0 && adj[last][next])
                                dp[next][mask | (1 << next)] += dp[last][mask];
            long ans = 0;
            for (int i = 0; i < n; i++) ans += dp[i][(1 << n) - 1];
            System.out.println(ans);
        }
    }
}
```

## **13. Good Trip (Problem 2049M)**

**Concept:**  
Expected value = sum of (friendship * probability of choosing both).

```java
import java.util.*;

public class GoodTrip {
    static long MOD = 1000000007;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            long m = sc.nextLong();
            long k = sc.nextLong();
            long sumF = 0;
            for (int i = 0; i < m; i++) {
                sc.nextLong(); sc.nextLong(); long f = sc.nextLong();
                sumF = (sumF + f) % MOD;
            }
            long totalPairs = n * (n - 1) / 2 % MOD;
            long invTotal = modInv(totalPairs);
            long prob = (k * (k - 1) / 2) % MOD * invTotal % MOD;
            long ans = sumF * prob % MOD;
            System.out.println(ans);
        }
    }
    private static long modInv(long a) {
        return pow(a, MOD - 2);
    }
    private static long pow(long a, long b) {
        long res = 1;
        while (b > 0) {
            if ((b & 1) == 1) res = res * a % MOD;
            a = a * a % MOD;
            b >>= 1;
        }
        return res;
    }
}
```

## **14. Blocking Elements (Problem 2049N)**

**Concept:**  
Binary search on max sum, greedy blocking.

```java
import java.util.*;

public class BlockingElements {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[] arr = new long[n];
            long total = 0;
            for (int i = 0; i < n; i++) {
                arr[i] = sc.nextLong();
                total += arr[i];
            }
            long l = 0, r = total;
            while (l < r) {
                long mid = (l + r) / 2;
                if (can(arr, mid)) r = mid;
                else l = mid + 1;
            }
            System.out.println(l);
        }
    }
    private static boolean can(long[] arr, long limit) {
        long sum = 0;
        for (long x : arr) {
            if (x > limit) return false;
            if (sum + x > limit) {
                sum = x;
            } else {
                sum += x;
            }
        }
        return true;
    }
}
```

```java
                curr = curr.child[c - 'a'];

        for (String w : words) {
            TrieNode curr = root;
            for (char c : w.toCharArray()) {

                ans += curr.count;
```

## **19. Tree XOR (Problem 2049S)**

```java
import java.util.*;

public class TreeXOR {

            for (int i = 1; i <= n; i++) tree[i] = new ArrayList<>();
            for (int i = 1; i <= n; i++) val[i] = sc.nextInt();
            for (int i = 0; i < n - 1; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                tree[u].add(v);
                tree[v].add(u);
            }
            ans = 0;
            dfs(1, -1);
            System.out.println(ans);
        }
    }
    private static int dfs(int node, int parent) {
        int xor = val[node];
        for (int child : tree[node]) if (child != parent) xor ^= dfs(child, node);
        ans += xor;
        return xor;
    }
```

```java
                    q.add(new int[]{v, nmask});
                }
            }
```

## **22. Fish Graph (Problem 2049V)**

**Concept:**  
DFS to find cycle + additional edge.

```java
import java.util.*;

public class FishGraph {
    static List<Integer>[] graph;
    static int[] parent;
    static boolean[] visited;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            for (int i = 0; i < m; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                graph[u].add(v);
                graph[v].add(u);
            }
            visited = new boolean[n + 1];
            parent = new int[n + 1];
            boolean found = false;
            for (int i = 1; i <= n && !found; i++) {
                if (!visited[i]) {
                    if (dfs(i, -1)) found = true;
                }
            }
            System.out.println(found ? "YES" : "NO");
        }
    }
    private static boolean dfs(int node, int par) {
        visited[node] = true;
        parent[node] = par;
        for (int child : graph[node]) {
            if (child == par) continue;
            if (visited[child]) {
                if (node != child) return true;
            } else {
                if (dfs(child, node)) return true;
            }
        }
        return false;
    }
}
```

## **23. Hot Start Up (easy version) (Problem 2049W)**

**Concept:**  
DP with last app running.

```java
import java.util.*;

public class HotStartUp {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int cold = sc.nextInt();
            int hot = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            long[][] dp = new long[n + 1][k + 1];
            for (int i = 0; i <= n; i++) Arrays.fill(dp[i], Long.MAX_VALUE);
            dp[0][0] = 0;
            for (int i = 0; i < n; i++) {
                for (int last = 0; last <= k; last++) {
                    if (dp[i][last] == Long.MAX_VALUE) continue;
                    int curr = arr[i];
                    long add = (curr == last) ? hot : cold;
                    dp[i + 1][curr] = Math.min(dp[i + 1][curr], dp[i][last] + add);
                    if (i == n - 1) continue;
                    int next = arr[i + 1];
                    if (curr == next) dp[i + 2][curr] = Math.min(dp[i + 2][curr], dp[i][last] + add + hot);
                    else dp[i + 2][next] = Math.min(dp[i + 2][next], dp[i][last] + add + cold);
                }
            }
            long ans = Long.MAX_VALUE;
            for (int last = 0; last <= k; last++) ans = Math.min(ans, dp[n][last]);
            System.out.println(ans);
        }
    }
}
```

## **25. Score of a Tree (Problem 2049Y)**

**Concept:**  
Score = sum over all paths of min value on path.

```java
import java.util.*;

public class ScoreOfATree {
    static List<Integer>[] tree;
    static int[] val;
    static long ans;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            tree = new ArrayList[n + 1];
            val = new int[n + 1];
            for (int i = 1; i <= n; i++) tree[i] = new ArrayList<>();
            for (int i = 1; i <= n; i++) val[i] = sc.nextInt();
            for (int i = 0; i < n - 1; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                tree[u].add(v);
                tree[v].add(u);
            }
            ans = 0;
            dfs(1, -1);
            System.out.println(ans);
        }
    }
    private static int dfs(int node, int parent) {
        int minVal = val[node];
        for (int child : tree[node]) if (child != parent) minVal = Math.min(minVal, dfs(child, node));
        ans += minVal;
        return minVal;
    }
}
```

## **26. Restore the Permutation (Problem 2049Z)**

**Concept:**  
Given b[i] = min(a[2i-1], a[2i]), restore lexicographically smallest a.

```java
import java.util.*;

public class RestorePermutation {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] b = new int[n];
            for (int i = 0; i < n; i++) b[i] = sc.nextInt();
            boolean[] used = new boolean[2 * n + 1];
            int[] a = new int[2 * n];
            int idx = 0;
            for (int i = 0; i < n; i++) {
                a[idx++] = b[i];
                used[b[i]] = true;
                int next = b[i] + 1;
                while (next <= 2 * n && used[next]) next++;
                if (next > 2 * n) {
                    idx = -1;
                    break;
                }
                a[idx++] = next;
                used[next] = true;
            }
            if (idx == -1) System.out.println(-1);
            else {
                for (int x : a) System.out.print(x + " ");
                System.out.println();
            }
        }
    }
}
```

## **29. Reset K Edges (Problem 2049C)**

**Concept:**  
Greedy: remove largest k edges.

```java
import java.util.*;

public class ResetKEdges {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            List<Integer>[] graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            for (int i = 0; i < n - 1; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                graph[u].add(v);
                graph[v].add(u);
            }
            int[] degree = new int[n + 1];
            for (int i = 1; i <= n; i++) degree[i] = graph[i].size();
            PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> b[0] - a[0]);
            for (int i = 1; i <= n; i++) if (degree[i] > 1) pq.add(new int[]{degree[i], i});
            int removed = 0;
            while (removed < k && !pq.isEmpty()) {
                int[] curr = pq.poll();
                removed++;
                degree[curr[1]]--;
                if (degree[curr[1]] > 1) pq.add(new int[]{degree[curr[1]], curr[1]});
            }
            int leaves = 0;
            for (int i = 1; i <= n; i++) if (degree[i] == 1) leaves++;
            System.out.println(leaves);
        }
    }
}
```

## **30. 2+ doors (Problem 2049D)**

**Concept:**  
Greedy assignment from smallest to largest.

```java
import java.util.*;

public class TwoPlusDoors {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] ans = new int[n + 1];
            List<int[]>[] constraints = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) constraints[i] = new ArrayList<>();
            for (int i = 0; i < m; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                int w = sc.nextInt();
                constraints[u].add(new int[]{v, w});
                constraints[v].add(new int[]{u, w});
            }
            for (int bit = 30; bit >= 0; bit--) {
                int[] need = new int[n + 1];
                for (int i = 1; i <= n; i++) {
                    for (int[] c : constraints[i]) {
                        if (((c[1] >> bit) & 1) == 0) {
                            need[i]++;
                        }
                    }
                }
                for (int i = 1; i <= n; i++) {
                    if (need[i] == constraints[i].size()) {
                        ans[i] |= (1 << bit);
                    }
                }
            }
            for (int i = 1; i <= n; i++) System.out.print(ans[i] + " ");
            System.out.println();
        }
    }
}
```

## **31. River Locks (Problem 2049E)**

**Concept:**  
Binary search on time, check if all can be filled.

```java
import java.util.*;

public class RiverLocks {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] cap = new long[n];
        for (int i = 0; i < n; i++) cap[i] = sc.nextLong();
        int q = sc.nextInt();
        while (q-- > 0) {
            long t = sc.nextLong();
            long l = 0, r = (long) 1e18;
            while (l < r) {
                long mid = (l + r) / 2;
                long water = mid;
                boolean ok = true;
                for (int i = 0; i < n; i++) {
                    if (water >= cap[i]) water -= cap[i];
                    else {
                        ok = false;
                        break;
                    }
                }
                if (ok) r = mid;
                else l = mid + 1;
            }
            System.out.println(l <= t ? l : -1);
        }
    }
}
```
