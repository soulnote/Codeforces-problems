

## **1. Gerrymandering (Problem 2005B)**

### **Problem Description:**
Divide grid into districts of equal size to maximize number of districts where one party wins.

### **Concept:**
- Greedy arrangement
- Try all possible district shapes

### **Java Code:**
```java
import java.util.*;

public class Gerrymandering {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();
            int ones = 0;
            for (char c : s.toCharArray()) if (c == '1') ones++;
            
            if (ones % 2 == 1 || ones == 0) {
                System.out.println("NO");
                continue;
            }
            
            int cnt = 0;
            for (int i = 0; i < n; i++) {
                if (s.charAt(i) == '1') cnt++;
                else cnt--;
                if (cnt < 0) break;
            }
            
            System.out.println(cnt >= 0 ? "YES" : "NO");
        }
    }
}
```

## **2. Rendez-vous de Marian et Robin (Problem 2005C)**

### **Problem Description:**
Find meeting point minimizing max distance for two people on graph.

### **Concept:**
- Multi-source BFS from both points
- Try meeting at each node

### **Java Code:**
```java
import java.util.*;

public class RendezVous {
    static List<int[]>[] graph;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int h = sc.nextInt();
            
            int[] horses = new int[h];
            for (int i = 0; i < h; i++) horses[i] = sc.nextInt();
            
            graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            
            for (int i = 0; i < m; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                int w = sc.nextInt();
                graph[u].add(new int[]{v, w});
                graph[v].add(new int[]{u, w});
            }
            
            long[] dist1 = dijkstra(1, horses);
            long[] distN = dijkstra(n, horses);
            
            long ans = Long.MAX_VALUE;
            for (int i = 1; i <= n; i++) {
                ans = Math.min(ans, Math.max(dist1[i], distN[i]));
            }
            System.out.println(ans == Long.MAX_VALUE ? -1 : ans);
        }
    }
    
    private static long[] dijkstra(int start, int[] horses) {
        int n = graph.length - 1;
        long[][] dist = new long[n + 1][2];
        for (int i = 1; i <= n; i++) Arrays.fill(dist[i], Long.MAX_VALUE);
        dist[start][0] = 0;
        
        PriorityQueue<long[]> pq = new PriorityQueue<>((a, b) -> Long.compare(a[0], b[0]));
        pq.add(new long[]{0, start, 0});
        
        Set<Integer> horseSet = new HashSet<>();
        for (int h : horses) horseSet.add(h);
        
        while (!pq.isEmpty()) {
            long[] curr = pq.poll();
            long d = curr[0];
            int u = (int) curr[1];
            int hasHorse = (int) curr[2];
            
            if (d > dist[u][hasHorse]) continue;
            
            if (horseSet.contains(u)) hasHorse = 1;
            long speed = hasHorse == 1 ? 1 : 2;
            
            for (int[] edge : graph[u]) {
                int v = edge[0];
                long w = edge[1];
                long nd = d + w / speed;
                if (nd < dist[v][hasHorse]) {
                    dist[v][hasHorse] = nd;
                    pq.add(new long[]{nd, v, hasHorse});
                }
            }
        }
        
        long[] res = new long[n + 1];
        for (int i = 1; i <= n; i++) res[i] = Math.min(dist[i][0], dist[i][1]);
        return res;
    }
}
```

- Greedy until budget exhausted

### **Java Code:**

```java
        int t = sc.nextInt();

            int n = sc.nextInt();
            long budget = sc.nextLong();
            int[][] items = new int[n][2];
            for (int i = 0; i < n; i++) items[i][0] = sc.nextInt();
            for (int i = 0; i < n; i++) items[i][1] = sc.nextInt();
            
            Arrays.sort(items, (a, b) -> {
                double r1 = (double) a[1] / a[0];
                double r2 = (double) b[1] / b[0];
                return Double.compare(r2, r1);
            });
            
            long ans = 0;
            for (int[] item : items) {
                long take = Math.min(budget / item[0], item[1]);
                ans += take;
                budget -= take * item[0];
            }
            System.out.println(ans);
        }
    }
```

## **6. Kim's Quest (Problem 2005G)**

### **Problem Description:**
Grid with treasures, find path collecting all with minimum turns.

### **Concept:**
- BFS with state (position, direction)

### **Java Code:**
```java
import java.util.*;

public class KimsQuest {
    static int[][] dirs = {{0,1},{1,0},{0,-1},{-1,0}};
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            char[][] grid = new char[n][m];
            for (int i = 0; i < n; i++) grid[i] = sc.next().toCharArray();
            
            int sr = 0, scPos = 0, er = 0, ec = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (grid[i][j] == 'S') { sr = i; scPos = j; }
                    if (grid[i][j] == 'E') { er = i; ec = j; }
                }
            }
            
            int[][][] dist = new int[n][m][4];
            for (int i = 0; i < n; i++) for (int j = 0; j < m; j++) Arrays.fill(dist[i][j], -1);
            
            PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> Integer.compare(a[3], b[3]));
            for (int d = 0; d < 4; d++) {
                dist[sr][scPos][d] = 0;
                pq.add(new int[]{sr, scPos, d, 0});
            }
            
            while (!pq.isEmpty()) {
                int[] curr = pq.poll();
                int r = curr[0], c = curr[1], dir = curr[2], turns = curr[3];
                if (turns > dist[r][c][dir]) continue;
                
                int nr = r + dirs[dir][0];
                int nc = c + dirs[dir][1];
                if (nr >= 0 && nr < n && nc >= 0 && nc < m && grid[nr][nc] != '#') {
                    if (dist[nr][nc][dir] == -1 || turns < dist[nr][nc][dir]) {
                        dist[nr][nc][dir] = turns;
                        pq.add(new int[]{nr, nc, dir, turns});
                    }
                }
                
                for (int nd = 0; nd < 4; nd++) {
                    if (nd == dir) continue;
                    if (dist[r][c][nd] == -1 || turns + 1 < dist[r][c][nd]) {
                        dist[r][c][nd] = turns + 1;
                        pq.add(new int[]{r, c, nd, turns + 1});
                    }
                }
            }
            
            int ans = Integer.MAX_VALUE;
            for (int d = 0; d < 4; d++) {
                if (dist[er][ec][d] != -1) ans = Math.min(ans, dist[er][ec][d]);
            }
            System.out.println(ans);
        }
    }
}
```

```java
                        x /= p;
                        cnt++;

                ans0 += (long) cnt * (cnt - 1) / 2;
                if (cnt % 2 == 0) ans1 += cnt / 2;
```

## **13. Recover an RBS (Problem 2005N)**

### **Problem Description:**
Bracket sequence with some positions unknown. Find if unique valid sequence exists.

### **Concept:**
- Greedy fill '(' first
- Check if swapping produces another valid

### **Java Code:**
```java
import java.util.*;

public class RecoverRBS {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            String s = sc.next();
            int n = s.length();
            int unknown = 0;
            int balance = 0;
            for (char c : s.toCharArray()) {
                if (c == '?') unknown++;
                else if (c == '(') balance++;
                else balance--;
            }
            
            int openNeeded = (n / 2 - balance) / 2;
            char[] arr = s.toCharArray();
            int openCount = 0;
            int lastOpen = -1;
            
            for (int i = 0; i < n; i++) {
                if (arr[i] == '?') {
                    if (openCount < openNeeded) {
                        arr[i] = '(';
                        openCount++;
                        lastOpen = i;
                    } else {
                        arr[i] = ')';
                    }
                }
            }
            
            if (lastOpen != -1) {
                arr[lastOpen] = ')';
                if (isValid(arr)) {
                    System.out.println("NO");
                } else {
                    System.out.println("YES");
                }
            } else {
                System.out.println("YES");
            }
        }
    }
    
    private static boolean isValid(char[] arr) {
        int bal = 0;
        for (char c : arr) {
            if (c == '(') bal++;
            else bal--;
            if (bal < 0) return false;
        }
        return bal == 0;
    }
}
```

## **14. Max GEQ Sum (Problem 2005O)**

### **Problem Description:**
Array, find maximum sum subarray where max >= sum.

### **Concept:**
- For each max, find left/right bounds
- Use monotonic stack

### **Java Code:**
```java
import java.util.*;

public class MaxGEQSum {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[] arr = new long[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextLong();
            
            long[] prefix = new long[n + 1];
            for (int i = 0; i < n; i++) prefix[i + 1] = prefix[i] + arr[i];
            
            int[] left = new int[n];
            int[] right = new int[n];
            Stack<Integer> stack = new Stack<>();
            
            for (int i = 0; i < n; i++) {
                while (!stack.isEmpty() && arr[stack.peek()] < arr[i]) stack.pop();
                left[i] = stack.isEmpty() ? 0 : stack.peek() + 1;
                stack.push(i);
            }
            
            stack.clear();
            for (int i = n - 1; i >= 0; i--) {
                while (!stack.isEmpty() && arr[stack.peek()] <= arr[i]) stack.pop();
                right[i] = stack.isEmpty() ? n - 1 : stack.peek() - 1;
                stack.push(i);
            }
            
            boolean possible = true;
            for (int i = 0; i < n; i++) {
                long maxSum = Math.max(prefix[right[i] + 1] - prefix[left[i]], 
                                       prefix[i + 1] - prefix[left[i]]);
                if (maxSum > arr[i]) {
                    possible = false;
                    break;
                }
            }
            System.out.println(possible ? "YES" : "NO");
        }
    }
}
```

## **15. Explorer Space (Problem 2005P)**

### **Problem Description:**
Grid with costs, find min cost to return to start after k steps.

### **Concept:**
- BFS layer by layer
- DP[k][i][j] = min cost

### **Java Code:**
```java
import java.util.*;

public class ExplorerSpace {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int k = sc.nextInt();
        int[][] right = new int[n][m - 1];
        int[][] down = new int[n - 1][m];
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m - 1; j++) {
                right[i][j] = sc.nextInt();
            }
        }
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < m; j++) {
                down[i][j] = sc.nextInt();
            }
        }
        
        if (k % 2 == 1) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    System.out.print(-1 + " ");
                }
                System.out.println();
            }
            return;
        }
        
        int[][][] dp = new int[k/2 + 1][n][m];
        for (int step = 1; step <= k/2; step++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    int min = Integer.MAX_VALUE;
                    if (i > 0) min = Math.min(min, dp[step - 1][i - 1][j] + down[i - 1][j]);
                    if (i < n - 1) min = Math.min(min, dp[step - 1][i + 1][j] + down[i][j]);
                    if (j > 0) min = Math.min(min, dp[step - 1][i][j - 1] + right[i][j - 1]);
                    if (j < m - 1) min = Math.min(min, dp[step - 1][i][j + 1] + right[i][j]);
                    dp[step][i][j] = min;
                }
            }
        }
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                System.out.print(dp[k/2][i][j] * 2 + " ");
            }
            System.out.println();
        }
    }
}
```

## **18. Road Reform (Problem 2005S)**

Roads with speeds, change speeds to make spanning tree with max-min ≤ k.

### **Concept:**
- Sort edges by speed
- Two pointers + DSU

### **Java Code:**
```java
import java.util.*;

public class RoadReform {
    static int[] parent;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int k = sc.nextInt();
            int[][] edges = new int[m][3];
            for (int i = 0; i < m; i++) {
                edges[i][0] = sc.nextInt() - 1;
                edges[i][1] = sc.nextInt() - 1;
                edges[i][2] = sc.nextInt();
            }
            Arrays.sort(edges, (a, b) -> Integer.compare(a[2], b[2]));
            
            parent = new int[n];
            for (int i = 0; i < n; i++) parent[i] = i;
            
            int ans = 0;
            int last = -1;
            for (int i = 0; i < m; i++) {
                if (union(edges[i][0], edges[i][1])) {
                    if (edges[i][2] > k) ans += edges[i][2] - k;
                    last = edges[i][2];
                }
            }
            
            if (last <= k) {
                int minDiff = Integer.MAX_VALUE;
                for (int[] e : edges) {
                    minDiff = Math.min(minDiff, Math.abs(e[2] - k));
                }
                ans = minDiff;
            }
            System.out.println(ans);
        }
    }
    
    private static int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }
    
    private static boolean union(int a, int b) {
        int pa = find(a);
        int pb = find(b);
        if (pa == pb) return false;
        parent[pa] = pb;
        return true;
    }
}
```

```java
    static List<Integer>[] tree;
    static long[] count;

        int t = sc.nextInt();

                for (int i = 0; i < m; i++) {
                    ans = (ans + primes[i] % MOD * (edgeCounts.get(i) % MOD)) % MOD;

                for (int i = 1; i < n - 1; i++) {
```
