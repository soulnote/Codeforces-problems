
## **1. Anna and the Valentine's Day Gift (Problem 1931A)**

### **Problem Description:**
Anna wants to find lexicographically smallest string of length 3 where sum of positions (a=1,b=2...) equals given number n.

### **Concept:**
- Greedy from left: place smallest possible character

### **Java Code:**
```java
import java.util.*;

public class AnnaValentineGift {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < 3; i++) {
                int maxVal = Math.min(26, n - (2 - i));
                sb.append((char)('a' + maxVal - 1));
                n -= maxVal;
            }
            System.out.println(sb.reverse());
        }
    }
}
```

## **3. Jumping Through Segments (Problem 1924A)**

### **Problem Description:**
Start at 0, jump k units, must land inside segments. Find if possible.

### **Concept:**
- Check parity of k and segment positions

### **Java Code:**
```java
import java.util.*;

public class JumpingThroughSegments {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int[] l = new int[n];
            int[] r = new int[n];
            for (int i = 0; i < n; i++) l[i] = sc.nextInt();
            for (int i = 0; i < n; i++) r[i] = sc.nextInt();
            
            int low = 0, high = (int)1e9;
            int ans = high;
            while (low <= high) {
                int mid = (low + high) / 2;
                long total = 0;
                for (int i = 0; i < n; i++) {
                    total += Math.max(0, r[i] - Math.max(l[i], mid) + 1);
                }
                if (total >= k) {
                    ans = mid;
                    low = mid + 1;
                } else {
                    high = mid - 1;
                }
            }
            System.out.println(ans);
        }
    }
}
```

```java
import java.util.*;
```

```java
                System.out.println(1);
                for (int i = 0; i < n; i++) System.out.print(1 + " ");
```

## **8. Copil Copac Draws Trees (Problem 1914D)**

### **Problem Description:**
Tree, draw edges in given order. Count good edges (edges that can be drawn without crossing).

### **Concept:**
- Process edges, track last drawn

### **Java Code:**
```java
import java.util.*;

public class CopilCopacDrawsTrees {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            List<Integer>[] graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            
            int[][] edges = new int[n - 1][2];
            for (int i = 0; i < n - 1; i++) {
                edges[i][0] = sc.nextInt();
                edges[i][1] = sc.nextInt();
                graph[edges[i][0]].add(i);
                graph[edges[i][1]].add(i);
            }
            
            int[] pos = new int[n + 1];
            Queue<Integer> q = new LinkedList<>();
            q.add(1);
            int ans = 0;
            
            while (!q.isEmpty()) {
                int u = q.poll();
                for (int idx : graph[u]) {
                    int v = (edges[idx][0] == u) ? edges[idx][1] : edges[idx][0];
                    if (pos[v] == 0 && v != 1) {
                        if (idx < pos[u]) ans++;
                        pos[v] = idx;
                        q.add(v);
                    }
                }
            }
            System.out.println(ans + 1);
        }
    }
}
```

## **11. Add Modulo 10 (Problem 1901B)**

### **Problem Description:**
Operation: a[i] = a[i] + a[i] % 10. Make all equal. Min operations.

### **Concept:**
- Numbers ending with 0 or 5 are special
- Others eventually cycle

### **Java Code:**
```java
import java.util.*;

public class AddModulo10 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            boolean has5or0 = false;
            for (int num : arr) {
                if (num % 10 == 5 || num % 10 == 0) has5or0 = true;
            }
            
            if (has5or0) {
                boolean possible = true;
                for (int i = 0; i < n; i++) {
                    if (arr[i] % 10 == 5) arr[i] += 5;
                    else if (arr[i] % 10 != 0) possible = false;
                }
                if (possible) {
                    for (int i = 1; i < n; i++) {
                        if (arr[i] != arr[0]) possible = false;
                    }
                }
                System.out.println(possible ? "Yes" : "No");
                continue;
            }
            
            int[] target = new int[n];
            for (int i = 0; i < n; i++) {
                while (arr[i] % 10 != 2) {
                    arr[i] += arr[i] % 10;
                }
                target[i] = arr[i] % 20;
            }
            
            boolean same = true;
            for (int i = 1; i < n; i++) {
                if (target[i] != target[0]) same = false;
            }
            System.out.println(same ? "Yes" : "No");
        }
    }
```

## **14. Weird Sum (Problem 1850D)**

### **Concept:**
- Separate x and y coordinates
- Sort and compute using prefix sums

### **Java Code:**
```java
import java.util.*;

public class WeirdSum {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        Map<Integer, List<Integer>> mapX = new HashMap<>();
        Map<Integer, List<Integer>> mapY = new HashMap<>();
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                int val = sc.nextInt();
                mapX.computeIfAbsent(val, k -> new ArrayList<>()).add(i);
                mapY.computeIfAbsent(val, k -> new ArrayList<>()).add(j);
            }
        }
        
        long ans = 0;
        for (List<Integer> list : mapX.values()) {
            Collections.sort(list);
            long prefix = 0;
            for (int i = 0; i < list.size(); i++) {
                ans += (long) list.get(i) * i - prefix;
                prefix += list.get(i);
            }
        }
        
        for (List<Integer> list : mapY.values()) {
            Collections.sort(list);
            long prefix = 0;
            for (int i = 0; i < list.size(); i++) {
                ans += (long) list.get(i) * i - prefix;
                prefix += list.get(i);
            }
        }
        
        System.out.println(ans);
    }
}
```

## **17. Berland Regional (Problem 1519C)**

### **Problem Description:**
Students from different universities with skills. For each k, form teams of size k with highest skills from each uni.

### **Concept:**
- Sort students within each uni
- Prefix sums for each uni
- For each uni, add to answer for k up to size

### **Java Code:**

```java
import java.util.*;

public class BerlandRegional {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] u = new int[n];
            int[] s = new int[n];
            for (int i = 0; i < n; i++) u[i] = sc.nextInt();
            for (int i = 0; i < n; i++) s[i] = sc.nextInt();
            
            Map<Integer, List<Long>> map = new HashMap<>();
            for (int i = 0; i < n; i++) {
                map.computeIfAbsent(u[i], k -> new ArrayList<>()).add((long) s[i]);
            }
            
            long[] ans = new long[n + 1];
            for (List<Long> list : map.values()) {
                list.sort(Collections.reverseOrder());
                long[] prefix = new long[list.size() + 1];
                for (int i = 0; i < list.size(); i++) {
                    prefix[i + 1] = prefix[i] + list.get(i);
                }
                for (int k = 1; k <= list.size(); k++) {
                    int maxTeams = list.size() / k;
                    ans[k] += prefix[maxTeams * k];
                }
            }
            
            for (int k = 1; k <= n; k++) {
                System.out.print(ans[k] + " ");
            }
            System.out.println();
        }
```

## **20. Zero Remainder Array (Problem 1374B)**

### **Problem Description:**
Array, operation: choose i, add 1 to a[i] if x not used, or add x then x++. Make all divisible by k.

### **Concept:**
- For each a[i], need (k - a[i] % k) % k additions
- Group by remainder, handle cycles

### **Java Code:**
```java
import java.util.*;

public class ZeroRemainderArray {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            Map<Integer, Integer> freq = new HashMap<>();
            int maxFreq = 0;
            int maxRem = 0;
            
            for (int i = 0; i < n; i++) {
                int x = sc.nextInt();
                int rem = x % k;
                if (rem == 0) continue;
                int need = k - rem;
                int cnt = freq.getOrDefault(need, 0) + 1;
                freq.put(need, cnt);
                if (cnt > maxFreq || (cnt == maxFreq && need > maxRem)) {
                    maxFreq = cnt;
                    maxRem = need;
                }
            }
            
            if (maxFreq == 0) {
                System.out.println(0);
            } else {
                System.out.println((long)(maxFreq - 1) * k + maxRem + 1);
            }
        }
    }
}
```

```java
                last = take;
            }

        }
    }
```

## **27. News Distribution (Problem 1167C)**

### **Problem Description:**
n users, m groups. Users in same group can share news. Find for each user, how many can receive.

### **Concept:**
- Union-Find / DSU
- Count size of each component

### **Java Code:**
```java
import java.util.*;

public class NewsDistribution {
    static int[] parent;
    static int[] size;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        parent = new int[n + 1];
        size = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            parent[i] = i;
            size[i] = 1;
        }
        
        for (int i = 0; i < m; i++) {
            int k = sc.nextInt();
            if (k == 0) continue;
            int first = sc.nextInt();
            for (int j = 1; j < k; j++) {
                int x = sc.nextInt();
                union(first, x);
            }
        }
        
        for (int i = 1; i <= n; i++) {
            System.out.print(size[find(i)] + " ");
        }
        System.out.println();
    }
    
    static int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }
    
    static void union(int a, int b) {
        int pa = find(a);
        int pb = find(b);
        if (pa != pb) {
            if (size[pa] < size[pb]) {
                int temp = pa;
                pa = pb;
                pb = temp;
            }
            parent[pb] = pa;
            size[pa] += size[pb];
        }
    }
}
```

## **29. Queen (Problem 1143C)**

### **Problem Description:**
Tree, each node has respect value. Remove nodes not respected by parent and all children don't respect it.

### **Concept:**

### **Java Code:**
```java
import java.util.*;

public class Queen {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] parent = new int[n + 1];
        int[] respect = new int[n + 1];
        
        for (int i = 1; i <= n; i++) {
            parent[i] = sc.nextInt();
            respect[i] = sc.nextInt();
        }
        
        List<Integer> result = new ArrayList<>();

            if (respect[i] == 0) continue;
            boolean allChildrenRespect = true;
            for (int j = 1; j <= n; j++) {

                    allChildrenRespect = false;
                    break;
                }
            }
            if (allChildrenRespect) result.add(i);
        }
        
        if (result.isEmpty()) {
            System.out.println(-1);
        } else {
            for (int v : result) System.out.print(v + " ");
            System.out.println();
        }
    }
```

## **30. Tape (Problem 1110B)**

n broken segments on tape of length m, can use k pieces of tape. Minimize total tape used.

- Remove largest gaps between segments

```java
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
```

## **31. Mashmokh and ACM (Problem 1106C)**

### **Problem Description:**
Count sequences of length k where each element divides next and all ≤ n.

### **Concept:**
- DP: dp[len][last] = count
- For each last, add to multiples

### **Java Code:**
```java
import java.util.*;

public class MashmokhAndACM {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        long MOD = 1000000007;
        
        long[][] dp = new long[k + 1][n + 1];
        Arrays.fill(dp[1], 1);
        
        for (int len = 1; len < k; len++) {
            for (int last = 1; last <= n; last++) {
                for (int mult = last; mult <= n; mult += last) {
                    dp[len + 1][mult] = (dp[len + 1][mult] + dp[len][last]) % MOD;
                }
            }
        }
        
        long ans = 0;
        for (int last = 1; last <= n; last++) {
            ans = (ans + dp[k][last]) % MOD;
        }
        System.out.println(ans);
    }
}
```
