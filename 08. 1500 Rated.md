## **5. Tea Tasting (Problem 2000B)**

### **Problem Description:**
n teas, each with volume. People drink sequentially. Find how much each person drinks.

### **Concept:**
- Prefix sums + binary search

### **Java Code:**
```java
import java.util.*;

public class TeaTasting {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[] a = new long[n];
            long[] b = new long[n];
            for (int i = 0; i < n; i++) a[i] = sc.nextLong();
            for (int i = 0; i < n; i++) b[i] = sc.nextLong();
            
            long[] prefix = new long[n + 1];
            for (int i = 0; i < n; i++) prefix[i + 1] = prefix[i] + b[i];
            
            long[] drink = new long[n];
            long[] count = new long[n];
            
            for (int i = 0; i < n; i++) {
                int l = i, r = n - 1;
                int idx = i - 1;
                while (l <= r) {
                    int mid = (l + r) / 2;
                    if (prefix[mid + 1] - prefix[i] <= a[i]) {
                        idx = mid;
                        l = mid + 1;
                    } else {
                        r = mid - 1;
                    }
                }
                if (idx >= i) {
                    count[i]++;
                    if (idx + 1 < n) count[idx + 1]--;
                    long remaining = a[i] - (prefix[idx + 1] - prefix[i]);
                    if (idx + 1 < n) drink[idx + 1] += remaining;
                } else {
                    drink[i] += a[i];
                }
            }
            
            for (int i = 1; i < n; i++) count[i] += count[i - 1];
            for (int i = 0; i < n; i++) {
                drink[i] += count[i] * b[i];
                System.out.print(drink[i] + " ");
            }
            System.out.println();
        }
    }
}
```

## **7. Palindrome Basis (Problem 1999C)**

### **Problem Description:**
Count ways to represent n as sum of palindromic numbers.

### **Concept:**
- DP, precompute palindromes up to 4e4

### **Java Code:**
```java
import java.util.*;

public class PalindromeBasis {
    static final int MOD = 1000000007;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        List<Integer> palindromes = new ArrayList<>();
        for (int i = 1; i <= 40000; i++) {
            if (isPalindrome(i)) palindromes.add(i);
        }
        
        int[] dp = new int[40001];
        dp[0] = 1;
        for (int p : palindromes) {
            for (int i = p; i <= 40000; i++) {
                dp[i] = (dp[i] + dp[i - p]) % MOD;
            }
        }
        
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            System.out.println(dp[n]);
        }
    }
    
    private static boolean isPalindrome(int n) {
        String s = String.valueOf(n);
        int l = 0, r = s.length() - 1;
        while (l < r) {
            if (s.charAt(l) != s.charAt(r)) return false;
            l++; r--;
        }
        return true;
    }
}
```

## **9. Factorials and Powers of Two (Problem 1999D)**

### **Problem Description:**
Represent n as sum of distinct factorials or powers of two. Minimize count.

### **Concept:**
- Precompute factorials
- Try all subsets of factorials, fill rest with powers of two

### **Java Code:**
```java
import java.util.*;

public class FactorialsAndPowers {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        List<Long> facts = new ArrayList<>();
        long fact = 1;
        for (int i = 1; fact <= 1000000000000L; i++) {
            fact *= i;
            facts.add(fact);
        }
        
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            int ans = Integer.MAX_VALUE;
            int m = facts.size();
            
            for (int mask = 0; mask < (1 << m); mask++) {
                long sum = 0;
                int count = 0;
                for (int i = 0; i < m; i++) {
                    if ((mask >> i & 1) == 1) {
                        sum += facts.get(i);
                        count++;
                    }
                }
                if (sum > n) continue;
                long remaining = n - sum;
                count += Long.bitCount(remaining);
                ans = Math.min(ans, count);
            }
            System.out.println(ans);
        }
    }
}
```

## **12. 13th Labour of Heracles (Problem 1996A)**

### **Problem Description:**
Tree with weights. Assign colors to edges. Value = sum over colors of (sum of nodes with that color). Maximize.

### **Concept:**
- Each node contributes weight * degree
- Sort node weights descending

### **Java Code:**
```java
import java.util.*;

public class LabourOfHeracles {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[] w = new long[n];
            for (int i = 0; i < n; i++) w[i] = sc.nextLong();
            
            int[] degree = new int[n];
            for (int i = 0; i < n - 1; i++) {
                int u = sc.nextInt() - 1;
                int v = sc.nextInt() - 1;
                degree[u]++;
                degree[v]++;
            }
            
            List<Long> contributions = new ArrayList<>();
            long sum = 0;
            for (int i = 0; i < n; i++) {
                sum += w[i];
                for (int j = 1; j < degree[i]; j++) {
                    contributions.add(w[i]);
                }
            }
            contributions.sort(Collections.reverseOrder());
            
            System.out.print(sum + " ");
            for (long c : contributions) {
                sum += c;
                System.out.print(sum + " ");
            }
            System.out.println();
        }
    }
}
```

## **14. k-Amazing Numbers (Problem 1995C)**

### **Problem Description:**
For each k from 1 to n, find smallest number that appears in every window of length k.

### **Concept:**
- For each value, track max gap between occurrences
- Update answer for k ≤ max_gap

### **Java Code:**
```java
import java.util.*;

public class KAmazingNumbers {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            Map<Integer, Integer> lastPos = new HashMap<>();
            Map<Integer, Integer> maxGap = new HashMap<>();
            
            for (int i = 0; i < n; i++) {
                int prev = lastPos.getOrDefault(arr[i], -1);
                int gap = i - prev;
                maxGap.put(arr[i], Math.max(maxGap.getOrDefault(arr[i], 0), gap));
                lastPos.put(arr[i], i);
            }
            
            for (int key : lastPos.keySet()) {
                int gap = n - lastPos.get(key);
                maxGap.put(key, Math.max(maxGap.get(key), gap));
            }
            
            int[] ans = new int[n + 2];
            Arrays.fill(ans, Integer.MAX_VALUE);
            
            for (Map.Entry<Integer, Integer> entry : maxGap.entrySet()) {
                int val = entry.getKey();
                int gap = entry.getValue();
                ans[gap] = Math.min(ans[gap], val);
            }
            
            for (int i = 1; i <= n; i++) {
                ans[i] = Math.min(ans[i], ans[i - 1]);
                System.out.print((ans[i] == Integer.MAX_VALUE ? -1 : ans[i]) + " ");
            }
            System.out.println();
        }
    }
}
```

## **19. Count Subrectangles (Problem 1995H)**

### **Problem Description:**
Binary matrix, count subrectangles with all 1s.

### **Concept:**
- For each row, count consecutive 1s
- For each column, use histogram method

### **Java Code:**
```java
import java.util.*;

public class CountSubrectangles {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[][] mat = new int[n][m];
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                mat[i][j] = sc.nextInt();
            }
        }
        
        int[] heights = new int[m];
        long ans = 0;
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (mat[i][j] == 1) heights[j]++;
                else heights[j] = 0;
            }
            
            Stack<Integer> stack = new Stack<>();
            int[] left = new int[m];
            int[] right = new int[m];
            
            for (int j = 0; j < m; j++) {
                while (!stack.isEmpty() && heights[stack.peek()] >= heights[j]) {
                    stack.pop();
                }
                left[j] = stack.isEmpty() ? 0 : stack.peek() + 1;
                stack.push(j);
            }
            
            stack.clear();
            for (int j = m - 1; j >= 0; j--) {
                while (!stack.isEmpty() && heights[stack.peek()] >= heights[j]) {
                    stack.pop();
                }
                right[j] = stack.isEmpty() ? m - 1 : stack.peek() - 1;
                stack.push(j);
            }
            
            for (int j = 0; j < m; j++) {
                ans += (long) heights[j] * (j - left[j] + 1) * (right[j] - j + 1);
            }
        }
        System.out.println(ans);
    }
}
```

- Use DSU on black edges

### **Java Code:**
```java
import java.util.*;

public class EdgyTrees {
    static int[] parent;
    static int[] size;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        parent = new int[n + 1];
        size = new int[n + 1];
        for (int i = 1; i <= n; i++) {
            parent[i] = i;
            size[i] = 1;
        }
        
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            int x = sc.nextInt();
            if (x == 0) union(u, v);
        }
        
        long MOD = 1000000007;
        long total = pow(n, k, MOD);
        long bad = 0;
        boolean[] seen = new boolean[n + 1];
        for (int i = 1; i <= n; i++) {
            int root = find(i);
            if (!seen[root]) {
                seen[root] = true;
                bad = (bad + pow(size[root], k, MOD)) % MOD;
            }
        }
        
        long ans = (total - bad + MOD) % MOD;
        System.out.println(ans);
    }
    
    static int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }
    
    static void union(int a, int b) {
        int pa = find(a);
        int pb = find(b);
        if (pa != pb) {
            if (size[pa] < size[pb]) {
                int temp = pa;
                pa = pb;
                pb = temp;
            }
            parent[pb] = pa;
            size[pa] += size[pb];
        }
    }
    
    static long pow(long a, long b, long mod) {
        long res = 1;
        while (b > 0) {
            if ((b & 1) == 1) res = res * a % mod;
            a = a * a % mod;
            b >>= 1;
        }
        return res;
    }
}
```

```java
        int n = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[n];

        Map<String, Integer> map = new HashMap<>();
        int zeroZero = 0;
        int zeroA = 0;
        
        for (int i = 0; i < n; i++) {

            else {

        System.out.println(max + zeroZero);

        while (b != 0) {
            long temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
```

```java
        List<Integer>[] graph = new ArrayList[n + 1];

            }
        }
        
        for (int v : result) System.out.print(v + " ");
        System.out.println();
```

```java
        int n = sc.nextInt();
```

```java
        int k2 = sc.nextInt();

            ans += val * val;
```
