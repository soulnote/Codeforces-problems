
## **1. Divisible Pairs (Problem 1931B)**

### **Problem Description:**
Count pairs (i, j) with i < j such that (a[i] + a[j]) is divisible by k.

### **Concept:**
- Use frequency of (a[i] % k)
- For remainder r, need remainder (k - r) % k

### **Java Code:**
```java
import java.util.*;

public class DivisiblePairs {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int[] freq = new int[k];
            for (int i = 0; i < n; i++) {
                int x = sc.nextInt();
                freq[x % k]++;
            }
            
            long ans = 0;
            ans += (long) freq[0] * (freq[0] - 1) / 2;
            for (int i = 1; i <= k / 2; i++) {
                if (i == k - i) {
                    ans += (long) freq[i] * (freq[i] - 1) / 2;
                } else {
                    ans += (long) freq[i] * freq[k - i];
                }
            }
            System.out.println(ans);
        }
    }
}
```

## **2. Find the Different Ones! (Problem 1927C)**

### **Problem Description:**
Two arrays, find if we can choose exactly k/2 elements from each such that all chosen elements are distinct and from 1 to k.

### **Concept:**
- Count elements present only in first, only in second, and in both
- Check if we can reach k/2 from each

### **Java Code:**
```java
import java.util.*;

public class FindDifferentOnes {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int k = sc.nextInt();
            
            Set<Integer> setA = new HashSet<>();
            Set<Integer> setB = new HashSet<>();
            
            for (int i = 0; i < n; i++) setA.add(sc.nextInt());
            for (int i = 0; i < m; i++) setB.add(sc.nextInt());
            
            int onlyA = 0, onlyB = 0, both = 0;
            for (int i = 1; i <= k; i++) {
                boolean inA = setA.contains(i);
                boolean inB = setB.contains(i);
                if (inA && inB) both++;
                else if (inA) onlyA++;
                else if (inB) onlyB++;
                else break;
            }
            
            if (onlyA <= k/2 && onlyB <= k/2 && onlyA + onlyB + both >= k) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }
    }
}
```

## **3. Romantic Glasses (Problem 1927D)**

### **Problem Description:**
Array, find if there exists subarray where sum of odd indices equals sum of even indices.

### **Concept:**
- Let b[i] = a[i] if i odd else -a[i]
- Then condition becomes subarray sum = 0
- Use prefix sums + set

### **Java Code:**
```java
import java.util.*;

public class RomanticGlasses {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[] arr = new long[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextLong();
            
            Set<Long> set = new HashSet<>();
            set.add(0L);
            long prefix = 0;
            boolean found = false;
            
            for (int i = 0; i < n; i++) {
                if (i % 2 == 0) prefix += arr[i];
                else prefix -= arr[i];
                
                if (set.contains(prefix)) {
                    found = true;
                    break;
                }
                set.add(prefix);
            }
            
            System.out.println(found ? "YES" : "NO");
        }
    }
```

## **4. Divide and Equalize (Problem 1881D)**

### **Problem Description:**
Array, operation: choose i, multiply a[i] by x, divide a[j] by x (must be divisible). Can make all equal?

### **Concept:**
- Prime factorize all numbers
- Total count of each prime must be divisible by n

### **Java Code:**
```java
import java.util.*;

public class DivideAndEqualize {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            Map<Integer, Integer> totalFreq = new HashMap<>();
            
            for (int i = 0; i < n; i++) {
                int x = sc.nextInt();
                for (int p = 2; p * p <= x; p++) {
                    while (x % p == 0) {
                        totalFreq.put(p, totalFreq.getOrDefault(p, 0) + 1);
                        x /= p;
                    }
                }
                if (x > 1) {
                    totalFreq.put(x, totalFreq.getOrDefault(x, 0) + 1);
                }
            }
            
            boolean possible = true;
            for (int count : totalFreq.values()) {
                if (count % n != 0) {
                    possible = false;
                    break;
                }
            }
            System.out.println(possible ? "YES" : "NO");
        }
    }
}
```

```java
                valid.add(sum);
                pow *= k;
                if (pow > 1000000000000L / k) break;
                sum += pow;
                if (sum > 1000000000000L) break;
            }
        }
```

## **10. Yet Another Problem About Pairs Satisfying an Inequality (Problem 1702C)**

### **Problem Description:**
Count pairs (i, j) with a[i] < i < j < a[j].

### **Concept:**
- Preprocess valid i's (a[i] < i)
- Preprocess valid j's (j < a[j])
- Binary search

### **Java Code:**
```java
import java.util.*;

public class PairsSatisfyingInequality {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            List<Integer> left = new ArrayList<>();
            List<Integer> right = new ArrayList<>();
            
            for (int i = 0; i < n; i++) {
                if (arr[i] < i + 1) left.add(i + 1);
                if (i + 1 < arr[i]) right.add(arr[i]);
            }
            
            Collections.sort(right);
            long ans = 0;
            for (int val : left) {
                int idx = upperBound(right, val);
                ans += right.size() - idx;
            }
            System.out.println(ans);
        }
    }
    
    private static int upperBound(List<Integer> list, int target) {
        int l = 0, r = list.size();
        while (l < r) {
            int mid = (l + r) / 2;
            if (list.get(mid) <= target) l = mid + 1;
            else r = mid;
        }
        return l;
    }
}
```

## **11. White-Black Balanced Subtrees (Problem 1676D)**

### **Problem Description:**
Tree with black/white nodes. Count subtrees with equal black and white.

### **Concept:**
- DFS, return balance (white - black) for each subtree

### **Java Code:**
```java
import java.util.*;

public class BalancedSubtrees {
    static List<Integer>[] tree;
    static String colors;
    static int count;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            tree = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) tree[i] = new ArrayList<>();
            
            for (int i = 2; i <= n; i++) {
                int p = sc.nextInt();
                tree[p].add(i);
            }
            colors = " " + sc.next();
            
            count = 0;
            dfs(1);
            System.out.println(count);
        }
    }
    
    private static int dfs(int node) {
        int balance = (colors.charAt(node) == 'W') ? 1 : -1;
        for (int child : tree[node]) {
            balance += dfs(child);
        }
        if (balance == 0) count++;
        return balance;
    }
}
```

## **12. Maximal AND (Problem 1669E)**

### **Problem Description:**
Array, k operations. Each operation: choose i, set a[i] = a[i] & (a[i] - 1) (remove lowest set bit). Maximize AND of all elements.

### **Concept:**
- Greedy from highest bit to lowest
- For each bit, check if we can set it in all numbers

### **Java Code:**
```java
import java.util.*;

public class MaximalAND {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            int ans = 0;
            for (int bit = 30; bit >= 0; bit--) {
                int need = 0;
                for (int num : arr) {
                    if (((num >> bit) & 1) == 0) need++;
                }
                if (need <= k) {
                    ans |= (1 << bit);
                    k -= need;
                }
            }
            System.out.println(ans);
        }
    }
}
```

Array, operation: choose k indices, replace each with XOR of all k. Find all k such that final array has all zeros.

### **Concept:**
- Final XOR of all elements = 0 for any k that works
- Check bitwise: each bit's count must be divisible by k

### **Java Code:**

```java
import java.util.*;

public class ArrayElimination {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

            int n = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            int xor = 0;
            for (int num : arr) xor ^= num;
            
            if (xor == 0) {
                System.out.println("YES");
                continue;
            }
            
            boolean found = false;
            for (int k = 2; k <= n; k++) {
                int target = 0;
                boolean valid = true;
                for (int i = 0; i < k; i++) target ^= arr[i];
                for (int i = k; i < n; i += k) {
                    int curr = 0;
                    for (int j = i; j < Math.min(n, i + k); j++) curr ^= arr[j];
                    if (curr != target) {
                        valid = false;
                        break;
                    }
                }
                if (valid) {
                    found = true;
                    break;
                }
            }
            System.out.println(found ? "YES" : "NO");
        }
    }
}
```

```java
            }

        }
    }
}
```

## **22. Buying Shovels (Problem 1360A)**

### **Problem Description:**
Need at least n shovels. Shop sells in packs of size k (1 to m). Min packs to buy.

### **Concept:**
- Find largest divisor of n that is ≤ m
- Answer = n / divisor

### **Java Code:**
```java
import java.util.*;

public class BuyingShovels {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            
            int ans = n;
            for (int i = 1; i * i <= n; i++) {
                if (n % i == 0) {
                    if (i <= m) ans = Math.min(ans, n / i);
                    if (n / i <= m) ans = Math.min(ans, i);
```

```java
            int n = sc.nextInt();
```

## **26. WOW Factor (Problem 1178A)**

### **Problem Description:**
String of 'v' and 'o', find number of "wow" patterns (v+v... + o + v+v...)

### **Concept:**
- Prefix vv count
- Suffix vv count
- For each 'o', add prefix * suffix

### **Java Code:**
```java
import java.util.*;

public class WowFactor {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        int n = s.length();
        
        int[] left = new int[n];
        int vCount = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'v') {
                vCount++;
            } else {
                vCount = 0;
            }
            left[i] = Math.max(0, vCount - 1);
            if (i > 0) left[i] += left[i - 1];
        }
        
        int[] right = new int[n];
        vCount = 0;
        for (int i = n - 1; i >= 0; i--) {
            if (s.charAt(i) == 'v') {
                vCount++;
            } else {
                vCount = 0;
            }
            right[i] = Math.max(0, vCount - 1);
            if (i < n - 1) right[i] += right[i + 1];
        }
        
        long ans = 0;
        for (int i = 0; i < n; i++) {
            if (s.charAt(i) == 'o') {
                long l = (i > 0) ? left[i - 1] : 0;
                long r = (i < n - 1) ? right[i + 1] : 0;
                ans += l * r;
            }
        }
        System.out.println(ans);
    }
}
```

- For given k, sort first k, pair largest with largest

### **Java Code:**

```java
import java.util.*;

public class NarrowFridge {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int h = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        int ans = 0;

            int mid = (l + r) / 2;
            int[] temp = Arrays.copyOfRange(arr, 0, mid);
            Arrays.sort(temp);

                ans = mid;
                l = mid + 1;
            } else {
                r = mid - 1;
            }
        }
        System.out.println(ans);
    }
```

```java
public class GoodArray {
```

## **29. Mahmoud and Ehab and the bipartiteness (Problem 862B)**

### **Problem Description:**
Tree, colored with 2 colors. Add max edges without breaking bipartite property.

### **Concept:**
- Count nodes in each color
- Max edges = color0 * color1 - (n - 1)

### **Java Code:**
```java
import java.util.*;

public class Bipartiteness {
    static List<Integer>[] graph;
    static int[] color;
    static long[] count = new long[2];
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        graph = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
        
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            graph[u].add(v);
            graph[v].add(u);
        }
        
        color = new int[n + 1];
        dfs(1, 0);
        
        long ans = count[0] * count[1] - (n - 1);
        System.out.println(ans);
    }
    
    private static void dfs(int node, int col) {
        color[node] = col;
        count[col]++;
        for (int child : graph[node]) {
            if (color[child] == -1) {
                dfs(child, 1 - col);
            }
        }
    }
}
```
