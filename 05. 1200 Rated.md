## **3. Plus Minus Permutation (Problem 1899B)**

### **Problem Description:**
Permutation of 1..n. Sum of positions divisible by x minus sum of positions divisible by y. Maximize.

### **Concept:**
- Put largest numbers on x-multiples
- Put smallest numbers on y-multiples
- Use arithmetic series

### **Java Code:**
```java
import java.util.*;

public class PlusMinusPermutation {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            long n = sc.nextLong();
            long x = sc.nextLong();
            long y = sc.nextLong();
            
            long lcm = x * y / gcd(x, y);
            long cntX = n / x;
            long cntY = n / y;
            long cntBoth = n / lcm;
            cntX -= cntBoth;
            cntY -= cntBoth;
            
            long sumX = (n + (n - cntX + 1)) * cntX / 2;
            long sumY = (1 + cntY) * cntY / 2;
            
            System.out.println(sumX - sumY);
        }
    }
    
    private static long gcd(long a, long b) {
        while (b != 0) {
            long temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

## **4. Assembly via Minimums (Problem 1857C)**

### **Problem Description:**
Given all pairwise minimums of an array, recover the original array.

### **Concept:**
- Sort the minimums array
- Original array = minimums[0], minimums[n-1], minimums[2n-3]...
- Last element is largest minimum + 1 or something

### **Java Code:**
```java
import java.util.*;

public class AssemblyViaMinimums {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = n * (n - 1) / 2;
            int[] b = new int[m];
            for (int i = 0; i < m; i++) b[i] = sc.nextInt();
            Arrays.sort(b);
            
            List<Integer> result = new ArrayList<>();
            int idx = 0;
            for (int i = n - 1; i >= 1; i--) {
                result.add(b[idx]);
                idx += i;
            }
            result.add(b[m - 1]);
            
            for (int num : result) System.out.print(num + " ");
            System.out.println();
        }
    }
}
```

## **5. Vika and the Bridge (Problem 1848A)**

### **Problem Description:**
Vika on grid, can change one plank color. Find minimum max gap.

### **Concept:**
- For each color, track gaps between positions
- Can reduce largest gap by half

### **Java Code:**
```java
import java.util.*;

public class VikaAndBridge {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int k = sc.nextInt();
            
            List<List<Integer>> positions = new ArrayList<>();
            for (int i = 0; i <= m; i++) positions.add(new ArrayList<>());
            
            for (int i = 0; i < n; i++) {
                int color = sc.nextInt();
                positions.get(color).add(i);
            }
            
            int ans = n;
            for (int color = 1; color <= m; color++) {
                List<Integer> pos = positions.get(color);
                if (pos.isEmpty()) continue;
                
                int first = pos.get(0);
                int last = n - 1 - pos.get(pos.size() - 1);
                
                List<Integer> gaps = new ArrayList<>();
                for (int i = 1; i < pos.size(); i++) {
                    gaps.add(pos.get(i) - pos.get(i - 1) - 1);
                }
                Collections.sort(gaps);
                
                int maxGap = gaps.isEmpty() ? 0 : gaps.get(gaps.size() - 1);
                int maxGap2 = gaps.size() >= 2 ? gaps.get(gaps.size() - 2) : 0;
                
                int best = Math.max(maxGap2, maxGap / 2);
                best = Math.max(best, Math.max(first, last));
                ans = Math.min(ans, best);
            }
            System.out.println(ans);
        }
    }
}
```

## **14. Mirror Grid (Problem 1702B)**

### **Problem Description:**
Grid of 0/1. Find min flips to make grid symmetric across both diagonals.

### **Concept:**
- Each position belongs to group of 4 symmetric positions
- For each group, take majority

### **Java Code:**
```java
import java.util.*;

public class MirrorGrid {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            char[][] grid = new char[n][n];
            for (int i = 0; i < n; i++) {
                grid[i] = sc.next().toCharArray();
            }
            
            int ans = 0;
            boolean[][] visited = new boolean[n][n];
            
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (visited[i][j]) continue;
                    
                    int ones = 0;
                    int x1 = i, y1 = j;
                    int x2 = j, y2 = n - 1 - i;
                    int x3 = n - 1 - i, y3 = n - 1 - j;
                    int x4 = n - 1 - j, y4 = i;
                    
                    ones += grid[x1][y1] - '0';
                    ones += grid[x2][y2] - '0';
                    ones += grid[x3][y3] - '0';
                    ones += grid[x4][y4] - '0';
                    
                    ans += Math.min(ones, 4 - ones);
                    
                    visited[x1][y1] = visited[x2][y2] = visited[x3][y3] = visited[x4][y4] = true;
                }
            }
            System.out.println(ans);
        }
    }
}
```

## **15. Binary Deque (Problem 1692C)**

### **Problem Description:**
Binary deque. Remove from ends to get sum s. Min removals.

### **Concept:**
- Sliding window to find longest subarray with sum s
- Answer = n - maxLen

### **Java Code:**
```java
import java.util.*;

public class BinaryDeque {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int s = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            int total = 0;
            for (int num : arr) total += num;
            
            if (total < s) {
                System.out.println(-1);
                continue;
            }
            if (total == s) {
                System.out.println(0);
                continue;
            }
            
            int needRemove = total - s;
            int left = 0, maxWindow = 0, currentZeros = 0;
            
            for (int right = 0; right < n; right++) {
                if (arr[right] == 0) currentZeros++;
                while (currentZeros > needRemove) {
                    if (arr[left] == 0) currentZeros--;
                    left++;
                }
                maxWindow = Math.max(maxWindow, right - left + 1);
            }
            
            System.out.println(n - maxWindow);
        }
    }
}
```

## **19. Make Them Equal (Problem 1627B)**

### **Problem Description:**
Array, can do operation: choose i and set a[i] = a[i] - x. Make all equal. Min operations.

### **Concept:**
- Find gcd of all differences
- Answer = sum of (max - a[i]) / gcd

### **Java Code:**
```java
import java.util.*;

public class MakeThemEqual {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[] arr = new long[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextLong();
            Arrays.sort(arr);
            
            long gcd = 0;
            for (int i = 1; i < n; i++) {
                long diff = arr[i] - arr[0];
                gcd = gcd(gcd, diff);
            }
            
            if (gcd == 0) {
                System.out.println(0);
                continue;
            }
            
            long ans = 0;
            for (int i = 0; i < n; i++) {
                ans += (arr[n - 1] - arr[i]) / gcd;
            }
            System.out.println(ans);
        }
    }
    
    private static long gcd(long a, long b) {
        while (b != 0) {
            long temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

## **20. Grandma Capa Knits a Scarf (Problem 1583A)**

### **Concept:**
- Try removing each letter
- Check if resulting string is palindrome

### **Java Code:**

```java
                        l++;
                    } else if (s.charAt(r) == c) {
                        removals++;
                        r--;
                    } else {
                        possible = false;
                        break;
                    }
```

### **Concept:**

- Or brute force for small a[i]

### **Java Code:**

```java
import java.util.*;

public class PleasantPairs {

        Scanner sc = new Scanner(System.in);

            int n = sc.nextInt();
            int[] arr = new int[n + 1];
            for (int i = 1; i <= n; i++) arr[i] = sc.nextInt();
            
            long ans = 0;
            for (int i = 1; i <= n; i++) {
                for (int j = arr[i] - i; j <= n; j += arr[i]) {
                    if (j > i && arr[i] * arr[j] == i + j) ans++;
                }
            }
            System.out.println(ans);
        }
    }
}
```

## **22. Stable Groups (Problem 1539A)**

### **Problem Description:**
Split array into groups where adjacent diff ≤ x. Min groups.

### **Concept:**
- Sort array
- Count gaps > x, each gap needs new group

### **Java Code:**
```java
import java.util.*;

public class StableGroups {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long x = sc.nextLong();
            Long[] arr = new Long[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextLong();
            Arrays.sort(arr);
            
            List<Long> gaps = new ArrayList<>();
            for (int i = 1; i < n; i++) {
                if (arr[i] - arr[i - 1] > x) {
                    gaps.add(arr[i] - arr[i - 1]);
                }
            }
            
            Collections.sort(gaps);
            int groups = gaps.size() + 1;
            long used = 0;
            
            for (long gap : gaps) {
                long need = (gap + x - 1) / x - 1;
                if (used + need <= groups - 1) {
                    used += need;
                    groups--;
                } else break;
            }
            
            System.out.println(groups);
        }
    }
}
```

## **27. Flip the Bits (Problem 1504B)**

### **Problem Description:**
Two binary strings a and b. Operation: choose prefix with equal 0s and 1s, flip bits. Can make a into b?

### **Concept:**
- Process from right to left
- Track flips parity
- Ensure at each prefix, 0s and 1s count equal when needed

### **Java Code:**
```java
import java.util.*;

public class FlipTheBits {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String a = sc.next();
            String b = sc.next();
            
            int balance = 0;
            boolean flipped = false;
            boolean possible = true;
            
            for (int i = n - 1; i >= 0; i--) {
                char ca = a.charAt(i);
                char cb = b.charAt(i);
                
                if (flipped) ca = (ca == '0') ? '1' : '0';
                
                if (ca != cb) {
                    if (balance == 0) {
                        possible = false;
                        break;
                    }
                }
                
                if (a.charAt(i) == '0') balance++;
                else balance--;
                
                if (balance == 0) flipped = !flipped;
            }
            
            System.out.println(possible ? "YES" : "NO");
        }
    }
}
```

## **28. M-Arrays (Problem 1490B)**

### **Problem Description:**
Array, you can rearrange. Want to minimize number of adjacent pairs with a[i] ≡ a[i+1] (mod m).

### **Concept:**
- Count frequencies of each remainder
- Greedy distribution

### **Java Code:**
```java
import java.util.*;

public class MArrays {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            int[] freq = new int[m];
            for (int i = 0; i < n; i++) {
                int x = sc.nextInt();
                freq[x % m]++;
            }
            
            int ans = 0;
            if (freq[0] > 0) ans++;
            
            for (int i = 1; i <= m / 2; i++) {
                if (i == m - i) {
                    if (freq[i] > 0) ans++;
                    continue;
                }
                int min = Math.min(freq[i], freq[m - i]);
                int max = Math.max(freq[i], freq[m - i]);
                if (min == 0 && max > 0) ans += max;
                else if (max == 0) continue;
                else {
                    ans += 1 + Math.max(0, max - min - 1);
                }
            }
            System.out.println(ans);
        }
    }
}
```
