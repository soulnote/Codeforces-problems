## **1. Partitioning the Array (Problem 1920B)**

### **Problem Description:**
Given array, partition into subarrays with equal sum. Find maximum number of parts.

### **Concept:**
- Total sum must be divisible by number of parts
- Check each divisor

### **Java Code:**
```java
import java.util.*;

public class PartitioningArray {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[] arr = new long[n];
            long total = 0;
            for (int i = 0; i < n; i++) {
                arr[i] = sc.nextLong();
                total += arr[i];
            }
            
            int ans = 1;
            for (int parts = n; parts >= 1; parts--) {
                if (total % parts == 0) {
                    long target = total / parts;
                    long sum = 0;
                    boolean ok = true;
                    for (int i = 0; i < n; i++) {
                        sum += arr[i];
                        if (sum > target) {
                            ok = false;
                            break;
                        }
                        if (sum == target) sum = 0;
                    }
                    if (ok && sum == 0) {
                        ans = parts;
                        break;
                    }
                }
            }
            System.out.println(ans);
        }
    }
}
```

## **3. Decreasing String (Problem 1920D)**

### **Problem Description:**
String s, repeatedly remove one character to make it decreasing. Find k-th string.

### **Concept:**
- Each removal creates new string
- Use combinatorial counting

### **Java Code:**
```java
import java.util.*;

public class DecreasingString {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long k = sc.nextLong();
            char[] s = sc.next().toCharArray();
            
            int[] last = new int[26];
            Arrays.fill(last, -1);
            for (int i = 0; i < n; i++) {
                last[s[i] - 'a'] = i;
            }
            
            boolean[] remove = new boolean[n];
            long total = 0;
            for (int c = 0; c < 26; c++) {
                if (last[c] == -1) continue;
                long cnt = 0;
                for (int i = 0; i <= last[c]; i++) {
                    if (!remove[i]) cnt++;
                }
                if (total + cnt < k) {
                    total += cnt;
                    for (int i = 0; i <= last[c]; i++) remove[i] = true;
                } else {
                    k -= total;
                    for (int i = 0; i <= last[c]; i++) {
                        if (!remove[i]) {
                            if (k == 1) {
                                System.out.print(s[i]);
                                break;
                            }
                            k--;
                        }
                    }
                    break;
                }
            }
            System.out.println();
        }
    }
}
```

## **6. Round Dance (Problem 1920G)**

### **Problem Description:**
n people in a circle, each knows two neighbors. Find number of possible circles.

### **Concept:**
- Each person has degree 2
- Count connected components

### **Java Code:**
```java
import java.util.*;

public class RoundDance {
    static List<Integer>[] graph;
    static boolean[] visited;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] a = new int[n + 1];
            for (int i = 1; i <= n; i++) a[i] = sc.nextInt();
            
            graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            
            for (int i = 1; i <= n; i++) {
                graph[i].add(a[i]);
                graph[a[i]].add(i);
            }
            
            visited = new boolean[n + 1];
            int cycles = 0, paths = 0;
            
            for (int i = 1; i <= n; i++) {
                if (!visited[i]) {
                    int size = dfs(i, -1);
                    if (size == 2) paths++;
                    else cycles++;
                }
            }
            
            System.out.println(cycles + " " + (cycles + paths));
        }
    }
    
    private static int dfs(int node, int parent) {
        visited[node] = true;
        int size = 1;
        for (int child : graph[node]) {
            if (child != parent && !visited[child]) {
                size += dfs(child, node);
            }
        }
        return size;
    }
}
```

## **7. Hits Different (Problem 1920H)**

### **Problem Description:**
Array, query (l, r). Count pairs (i, j) with i < j and a[i] * a[j] divisible by k.

### **Concept:**
- Use frequency of gcd with k

### **Java Code:**
```java
import java.util.*;

public class HitsDifferent {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
        
        int[] factors = new int[k + 1];
        for (int i = 1; i <= k; i++) {
            if (k % i == 0) factors[i] = 1;
        }
        
        int q = sc.nextInt();
        while (q-- > 0) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            Map<Integer, Integer> freq = new HashMap<>();
            long ans = 0;
            for (int i = l - 1; i < r; i++) {
                int gcd = gcd(arr[i], k);
                int need = k / gcd;
                ans += freq.getOrDefault(need, 0);
                freq.put(gcd, freq.getOrDefault(gcd, 0) + 1);
            }
            System.out.println(ans);
        }
    }
    
    private static int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

```java
                for (int i = 1; i < neg.size(); i++) result.add(neg.get(i));

                result.addAll(neg);
                result.addAll(zero);
                result.addAll(pos);
```

## **10. Equal Frequencies (Problem 1920K)**

### **Problem Description:**
String, change minimum characters so all characters have same frequency.

### **Concept:**
- Try all possible frequencies
- Greedy replacement

### **Java Code:**
```java
import java.util.*;

public class EqualFrequencies {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();
            int[] freq = new int[26];
            for (char c : s.toCharArray()) freq[c - 'a']++;
            
            int bestFreq = 0;
            int bestDistinct = 0;
            for (int d = 1; d <= 26; d++) {
                if (n % d == 0) {
                    int target = n / d;
                    int cnt = 0;
                    for (int f : freq) cnt += Math.min(f, target);
                    if (cnt >= target * d) {
                        bestFreq = target;
                        bestDistinct = d;
                    }
                }
            }
            
            char[] result = s.toCharArray();
            List<Integer> over = new ArrayList<>();
            List<Integer> under = new ArrayList<>();
            for (int i = 0; i < 26; i++) {
                if (freq[i] > bestFreq) over.add(i);
                else if (freq[i] < bestFreq) under.add(i);
            }
            
            int oi = 0, ui = 0;
            for (int i = 0; i < n; i++) {
                int idx = result[i] - 'a';
                if (freq[idx] > bestFreq) {
                    if (under.size() > ui) {
                        freq[idx]--;
                        result[i] = (char)('a' + under.get(ui));
                        freq[under.get(ui)]++;
                        if (freq[under.get(ui)] == bestFreq) ui++;
                    }
                }
            }
            
            System.out.println(new String(result));
        }
    }
}
```

```java
public class SplitIntoSets {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

            int n = sc.nextInt();

            }

            for (int i = 0; i < n; i++) {
                graph[a[i]].add(b[i]);
                graph[b[i]].add(a[i]);
            }
            
            int[] color = new int[n + 1];
            Arrays.fill(color, -1);

                    q.add(i);

                            if (color[v] == -1) {
                                color[v] = color[u] ^ 1;
                                q.add(v);
                            } else if (color[v] == color[u]) {
                                possible = false;
                            }
                        }
                    }
                }
            }
            
            System.out.println(possible ? "YES" : "NO");
        }
    }
```

### **Concept:**
- Only strings with alternating pattern work
- Try 6 patterns

### **Java Code:**
```java
import java.util.*;

public class SayNoToPalindromes {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        String s = sc.next();
        
        String[] patterns = {"abc", "acb", "bac", "bca", "cab", "cba"};
        int[][] cost = new int[6][n + 1];
        
        for (int p = 0; p < 6; p++) {
            for (int i = 0; i < n; i++) {
                cost[p][i + 1] = cost[p][i] + (s.charAt(i) == patterns[p].charAt(i % 3) ? 0 : 1);
            }
        }
        
        while (m-- > 0) {
            int l = sc.nextInt();
            int r = sc.nextInt();
            int ans = Integer.MAX_VALUE;
            for (int p = 0; p < 6; p++) {
                ans = Math.min(ans, cost[p][r] - cost[p][l - 1]);
            }
            System.out.println(ans);
        }
    }
}
```

## **22. Parsa's Humongous Tree (Problem 1920W)**

### **Problem Description:**
Tree, each node can choose value in [l[i], r[i]]. Maximize sum of |a[u] - a[v]| over edges.

### **Concept:**
- DP: at each node, value is either l[i] or r[i]
- Tree DP

### **Java Code:**
```java
import java.util.*;

public class ParsaTree {
    static List<Integer>[] tree;
    static long[][] dp;
    static long[] l, r;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            l = new long[n + 1];
            r = new long[n + 1];
            for (int i = 1; i <= n; i++) {
                l[i] = sc.nextLong();
                r[i] = sc.nextLong();
            }
            
            tree = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) tree[i] = new ArrayList<>();
            
            for (int i = 0; i < n - 1; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                tree[u].add(v);
                tree[v].add(u);
            }
            
            dp = new long[n + 1][2];
            dfs(1, -1);
            System.out.println(Math.max(dp[1][0], dp[1][1]));
        }
    }
    
    private static void dfs(int node, int parent) {
        for (int child : tree[node]) {
            if (child == parent) continue;
            dfs(child, node);
            dp[node][0] += Math.max(dp[child][0] + Math.abs(l[node] - l[child]), dp[child][1] + Math.abs(l[node] - r[child]));
            dp[node][1] += Math.max(dp[child][0] + Math.abs(r[node] - l[child]), dp[child][1] + Math.abs(r[node] - r[child]));
        }
    }
}
```

```java
        Scanner sc = new Scanner(System.in);

        long MOD = 1000000007;
        
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            
            long[][] dp = new long[k + 1][2];
            dp[k][0] = 1;

                long[][] ndp = new long[k + 1][2];
                for (int j = 1; j <= k; j++) {

                    ndp[j][0] = (ndp[j][0] + dp[j][0]) % MOD;

                }
                dp = ndp;
            }
            
            long ans = 0;
            for (int j = 1; j <= k; j++) {
                ans = (ans + dp[j][0] + dp[j][1]) % MOD;
            }
```

## **25. Row GCD (Problem 1920Z)**

### **Problem Description:**
Two arrays, find GCD of (a[i] + b[j]) for all i, j.

### **Concept:**
- GCD(a[0]+b[j], a[1]+b[j], ...) = GCD(a[0]+b[j], diff1, diff2, ...)
- Where diff = a[i] - a[0]

### **Java Code:**
```java
import java.util.*;

public class RowGCD {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        long[] a = new long[n];
        long[] b = new long[m];
        for (int i = 0; i < n; i++) a[i] = sc.nextLong();
        for (int i = 0; i < m; i++) b[i] = sc.nextLong();
        
        long diffGcd = 0;
        for (int i = 1; i < n; i++) {
            diffGcd = gcd(diffGcd, Math.abs(a[i] - a[0]));
        }
        
        for (int i = 0; i < m; i++) {
            System.out.print(gcd(a[0] + b[i], diffGcd) + " ");
        }
        System.out.println();
    }
    
    private static long gcd(long a, long b) {
        while (b != 0) {
            long temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

## **28. Array Walk (Problem 1920C)**

### **Problem Description:**
Array, start at 1, can move right or left (max k left moves). Maximize sum.

### **Concept:**
- Try each number of left moves
- Find max sum in window

### **Java Code:**
```java
import java.util.*;

public class ArrayWalk {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int z = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            long[] prefix = new long[n + 1];
            for (int i = 0; i < n; i++) prefix[i + 1] = prefix[i] + arr[i];
            
            long ans = 0;
            for (int left = 0; left <= z; left++) {
                int right = k - 2 * left;
                if (right < 0) continue;
                int pos = right;
                long sum = prefix[pos + 1];
                long maxPair = 0;
                for (int i = 0; i < pos; i++) {
                    maxPair = Math.max(maxPair, arr[i] + arr[i + 1]);
                }
                sum += maxPair * left;
                ans = Math.max(ans, sum);
            }
            System.out.println(ans);
        }
    }
}
```

## **29. Orac and LCM (Problem 1920D)**

### **Problem Description:**
Array, take LCM of all pairs, then GCD of those LCMs.

### **Concept:**
- For each prime, take second smallest exponent

### **Java Code:**
```java
import java.util.*;

public class OracAndLCM {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        long[] arr = new long[n];
        for (int i = 0; i < n; i++) arr[i] = sc.nextLong();
        
        long[] suffix = new long[n + 1];
        for (int i = n - 1; i >= 0; i--) {
            suffix[i] = gcd(suffix[i + 1], arr[i]);
        }
        
        long ans = 0;
        long prefix = 0;
        for (int i = 0; i < n - 1; i++) {
            prefix = gcd(prefix, arr[i]);
            ans = gcd(ans, lcm(prefix, suffix[i + 1]));
        }
        System.out.println(ans);
    }
    
    private static long gcd(long a, long b) {
        while (b != 0) {
            long temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
    
    private static long lcm(long a, long b) {
        return a / gcd(a, b) * b;
    }
}
```

## **30. Linova and Kingdom (Problem 1920E)**

### **Problem Description:**
Tree, choose k cities as industrial. Sum of distances from each industrial to nearest tourist.

### **Concept:**
- Sort nodes by (depth - subtree_size)
- Choose largest

### **Java Code:**
```java
import java.util.*;

public class LinovaAndKingdom {
    static List<Integer>[] tree;
    static long[] depth;
    static int[] size;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        tree = new ArrayList[n + 1];
        for (int i = 1; i <= n; i++) tree[i] = new ArrayList<>();
        
        for (int i = 0; i < n - 1; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            tree[u].add(v);
            tree[v].add(u);
        }
        
        depth = new long[n + 1];
        size = new int[n + 1];
        dfs(1, -1, 0);
        
        Long[] val = new Long[n];
        for (int i = 1; i <= n; i++) {
            val[i - 1] = depth[i] - (size[i] - 1);
        }
        Arrays.sort(val, Collections.reverseOrder());
        
        long ans = 0;
        for (int i = 0; i < k; i++) ans += val[i];
        System.out.println(ans);
    }
    
    private static void dfs(int node, int parent, long d) {
        depth[node] = d;
        size[node] = 1;
        for (int child : tree[node]) {
            if (child != parent) {
                dfs(child, node, d + 1);
                size[node] += size[child];
            }
        }
    }
}
```
