
## **1. Maximum Modulo Equality (Problem 2004B)**

### **Problem Description:**
Find maximum x such that all elements modulo x are equal.

### **Concept:**
- All numbers must be congruent modulo x
- x must divide all differences

### **Java Code:**
```java
import java.util.*;

public class MaximumModuloEquality {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long[] arr = new long[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextLong();
            Arrays.sort(arr);
            
            long gcd = 0;
            for (int i = 1; i < n; i++) {
                gcd = gcd(gcd, arr[i] - arr[0]);
            }
            System.out.println(gcd == 0 ? n : gcd);
        }
    }
    
    private static long gcd(long a, long b) {
        while (b != 0) {
            long temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

## **2. Drunken Maze (Problem 2004C)**

### **Problem Description:**
Maze with walls, find path with maximum sum avoiding obstacles.

### **Concept:**
- BFS/DFS with state (position, direction)

### **Java Code:**
```java
import java.util.*;

public class DrunkenMaze {
    static int[][] dirs = {{0,1},{1,0},{0,-1},{-1,0}};
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            char[][] grid = new char[n][m];
            for (int i = 0; i < n; i++) {
                grid[i] = sc.next().toCharArray();
            }
            
            int sr = 0, scPos = 0;
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (grid[i][j] == 'L') {
                        sr = i;
                        scPos = j;
                    }
                }
            }
            
            boolean[][] visited = new boolean[n][m];
            Queue<int[]> q = new LinkedList<>();
            q.add(new int[]{sr, scPos});
            visited[sr][scPos] = true;
            
            while (!q.isEmpty()) {
                int[] curr = q.poll();
                int r = curr[0], c = curr[1];
                for (int[] d : dirs) {
                    int nr = r + d[0], nc = c + d[1];
                    if (nr >= 0 && nr < n && nc >= 0 && nc < m && grid[nr][nc] != '#' && !visited[nr][nc]) {
                        int cnt = 0;
                        for (int[] dd : dirs) {
                            int nnr = nr + dd[0], nnc = nc + dd[1];
                            if (nnr >= 0 && nnr < n && nnc >= 0 && nnc < m && grid[nnr][nnc] != '#' && !visited[nnr][nnc]) {
                                cnt++;
                            }
                        }
                        if (cnt <= 1) {
                            visited[nr][nc] = true;
                            q.add(new int[]{nr, nc});
                        }
                    }
                }
            }
            
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    if (grid[i][j] == '.' && visited[i][j]) System.out.print('+');
                    else System.out.print(grid[i][j]);
                }
                System.out.println();
            }
        }
    }
}
```

## **3. Tree Pruning (Problem 2004D)**

### **Problem Description:**
Tree, remove nodes to minimize height while keeping k leaves.

### **Concept:**
- Binary search on height
- Greedy removal

### **Java Code:**
```java
import java.util.*;

public class TreePruning {
    static List<Integer>[] tree;
    static int[] depth;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            tree = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) tree[i] = new ArrayList<>();
            
            for (int i = 0; i < n - 1; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                tree[u].add(v);
                tree[v].add(u);
            }
            
            int l = 1, r = n, ans = 0;
            while (l <= r) {
                int mid = (l + r) / 2;
                if (check(1, -1, mid) >= k) {
                    ans = mid;
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }
            System.out.println(ans);
        }
    }
    
    private static int check(int node, int parent, int limit) {
        int leaves = 0;
        int maxDepth = 0;
        for (int child : tree[node]) {
            if (child != parent) {
                int childVal = check(child, node, limit);
                if (childVal >= limit) leaves++;
                else maxDepth = Math.max(maxDepth, childVal);
            }
        }
        return leaves + (maxDepth + 1);
    }
}
```

```java
            }
        }
        return inv;
```

## **9. Chat Screenshots (Problem 2004J)**

### **Problem Description:**
n users, m screenshots each showing k users in order. Find if consistent.

### **Concept:**
- Build graph of order constraints
- Check for cycle

### **Java Code:**
```java
import java.util.*;

public class ChatScreenshots {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            List<Integer>[] graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            
            int[] indeg = new int[n + 1];
            
            for (int i = 0; i < m; i++) {
                int k = sc.nextInt();
                int prev = -1;
                for (int j = 0; j < k; j++) {
                    int x = sc.nextInt();
                    if (prev != -1 && prev != x) {
                        graph[prev].add(x);
                    }
                    prev = x;
                }
            }
            
            // Remove duplicates and count indegrees
            for (int i = 1; i <= n; i++) {
                Set<Integer> set = new HashSet<>(graph[i]);
                graph[i] = new ArrayList<>(set);
                for (int v : graph[i]) indeg[v]++;
            }
            
            Queue<Integer> q = new LinkedList<>();
            for (int i = 1; i <= n; i++) {
                if (indeg[i] == 0) q.add(i);
            }
            
            int cnt = 0;
            while (!q.isEmpty()) {
                int u = q.poll();
                cnt++;
                for (int v : graph[u]) {
                    indeg[v]--;
                    if (indeg[v] == 0) q.add(v);
                }
            }
            
            System.out.println(cnt == n ? "YES" : "NO");
        }
    }
}
```

- Use prefix XOR counts

```java
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

            int[] cnt = new int[2];

}
```

## **12. Ira and Flamenco (Problem 2004M)**

### **Problem Description:**
Array, choose m elements with distinct values. Count ways such that max - min < m.

### **Concept:**
- Sort array
- Two pointers + combinatorics

### **Java Code:**
```java
import java.util.*;

public class IraAndFlamenco {
    static long MOD = 1000000007;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            Integer[] arr = new Integer[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            Arrays.sort(arr);
            
            long ans = 0;
            int j = 0;
            Map<Integer, Integer> freq = new HashMap<>();
            for (int i = 0; i < n; i++) {
                freq.put(arr[i], freq.getOrDefault(arr[i], 0) + 1);
                while (arr[i] - arr[j] >= m) {
                    freq.put(arr[j], freq.get(arr[j]) - 1);
                    if (freq.get(arr[j]) == 0) freq.remove(arr[j]);
                    j++;
                }
                if (freq.size() == m) {
                    long ways = 1;
                    for (int v : freq.values()) ways = ways * v % MOD;
                    ans = (ans + ways) % MOD;
                }
            }
            System.out.println(ans);
        }
    }
}
```

## **13. Don't Blame Me (Problem 2004N)**

### **Problem Description:**
Array, count subsequences where AND of all elements equals 0.

### **Concept:**
- DP with bitmask (max 64 states)

### **Java Code:**
```java
import java.util.*;

public class DontBlameMe {
    static long MOD = 1000000007;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            long[] dp = new long[64];
            dp[63] = 1;
            
            for (int num : arr) {
                long[] ndp = dp.clone();
                for (int mask = 0; mask < 64; mask++) {
                    ndp[mask & num] = (ndp[mask & num] + dp[mask]) % MOD;
                }
                dp = ndp;
            }
            
            long ans = 0;
            for (int mask = 0; mask < (1 << k); mask++) {
                ans = (ans + dp[mask]) % MOD;
            }
            System.out.println(ans);
        }
    }
}
```

```java
import java.util.*;

public class FixedPrefixPermutations {
    static class TrieNode {
        Map<Integer, TrieNode> children = new HashMap<>();
        int count = 0;
    }
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

            int n = sc.nextInt();
            int m = sc.nextInt();
            int[][] perms = new int[n][m];
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {
                    perms[i][j] = sc.nextInt();

            for (int i = 0; i < n; i++) {
                TrieNode curr = root;
                for (int j = 0; j < m; j++) {
                    int val = perms[i][j];
                    if (!curr.children.containsKey(val)) {
                        curr.children.put(val, new TrieNode());
                    }
                    curr = curr.children.get(val);
                    curr.count++;
                }

            int[] ans = new int[n];
            for (int i = 0; i < n; i++) {
                TrieNode curr = root;
                int len = 0;
                for (int j = 0; j < m; j++) {
                    int val = perms[i][j];
                    if (curr.children.containsKey(val)) {
                        curr = curr.children.get(val);
                        len++;
                    } else break;
                }
                ans[i] = len;
            }
            
            for (int x : ans) System.out.print(x + " ");
            System.out.println();
        }
```

## **18. SlavicG's Favorite Problem (Problem 2004S)**

### **Problem Description:**
Tree with edge weights, find if path from a to b has XOR = 0 after one teleport.

### **Concept:**
- Store XOR from a to all nodes
- Check if any node has same XOR from b

### **Java Code:**
```java
import java.util.*;

public class SlavicGFavorite {
    static List<int[]>[] graph;
    static Set<Integer> set;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int a = sc.nextInt();
            int b = sc.nextInt();
            graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            
            for (int i = 0; i < n - 1; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                int w = sc.nextInt();
                graph[u].add(new int[]{v, w});
                graph[v].add(new int[]{u, w});
            }
            
            set = new HashSet<>();
            dfs(a, -1, 0, b);
            
            boolean ok = dfs2(b, -1, 0);
            System.out.println(ok ? "YES" : "NO");
        }
    }
    
    private static void dfs(int node, int parent, int xor, int avoid) {
        if (node != avoid) set.add(xor);
        for (int[] edge : graph[node]) {
            int child = edge[0];
            int w = edge[1];
            if (child != parent && child != avoid) {
                dfs(child, node, xor ^ w, avoid);
            }
        }
    }
    
    private static boolean dfs2(int node, int parent, int xor) {
        if (node != parent && set.contains(xor)) return true;
        for (int[] edge : graph[node]) {
            int child = edge[0];
            int w = edge[1];
            if (child != parent) {
                if (dfs2(child, node, xor ^ w)) return true;
            }
        }
        return false;
    }
}
```

## **21. Monoblock (Problem 2004V)**

### **Problem Description:**
Count number of maximal constant segments after operations.

### **Concept:**
- Each change in value creates new block

### **Java Code:**
```java
import java.util.*;

public class Monoblock {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        int[] arr = new int[n + 2];
        for (int i = 1; i <= n; i++) arr[i] = sc.nextInt();
        
        long ans = 0;
        for (int i = 1; i <= n; i++) {
            ans += (long) i * (n - i + 1);
        }
        
        for (int i = 2; i <= n; i++) {
            if (arr[i] == arr[i - 1]) {
                ans -= (long) (i - 1) * (n - i + 1);
            }
        }
        
        while (m-- > 0) {
            int idx = sc.nextInt();
            int val = sc.nextInt();
            
            if (idx > 1 && arr[idx] == arr[idx - 1]) ans += (long) (idx - 1) * (n - idx + 1);
            if (idx < n && arr[idx] == arr[idx + 1]) ans += (long) idx * (n - idx);
            
            arr[idx] = val;
            
            if (idx > 1 && arr[idx] == arr[idx - 1]) ans -= (long) (idx - 1) * (n - idx + 1);
            if (idx < n && arr[idx] == arr[idx + 1]) ans -= (long) idx * (n - idx);
            
            System.out.println(ans);
        }
    }
}
```

```java
public class Gambling {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int[] arr = new int[n];
            for (int i = 0; i < n; i++) arr[i] = sc.nextInt();
            
            Map<Integer, List<Integer>> pos = new HashMap<>();
            for (int i = 0; i < n; i++) {
                pos.computeIfAbsent(arr[i], k -> new ArrayList<>()).add(i);
            }
            
            int bestNum = arr[0];

                        bestVal = sum;
                        bestNum = num;
                        bestL = l + 1;
                        bestR = p.get(i) + 1;
                    }
                    if (sum < 1) {
                        sum = 1;
                        l = p.get(i);
                    }
                }
            }
            System.out.println(bestNum + " " + bestL + " " + bestR);
        }
    }
```

## **26. Road Optimization (Problem 2004A)**

### **Problem Description:**
n road segments with speed limits. Remove k signs to minimize travel time.

### **Concept:**
- DP[i][j] = min time using first i signs, removing j

### **Java Code:**
```java
import java.util.*;

public class RoadOptimization {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int l = sc.nextInt();
        int k = sc.nextInt();
        int[] d = new int[n + 1];
        int[] a = new int[n];
        for (int i = 0; i < n; i++) d[i] = sc.nextInt();
        for (int i = 0; i < n; i++) a[i] = sc.nextInt();
        d[n] = l;
        
        long[][] dp = new long[n + 1][k + 1];
        for (int i = 0; i <= n; i++) Arrays.fill(dp[i], Long.MAX_VALUE);
        dp[0][0] = 0;
        
        for (int i = 1; i <= n; i++) {
            for (int j = 0; j <= k; j++) {
                for (int last = i - 1; last >= 0; last--) {
                    int removed = (i - 1) - last;
                    if (j >= removed && dp[last][j - removed] != Long.MAX_VALUE) {
                        dp[i][j] = Math.min(dp[i][j], dp[last][j - removed] + a[last] * (d[i] - d[last]));
                    }
                }
            }
        }
        
        long ans = Long.MAX_VALUE;
        for (int j = 0; j <= k; j++) ans = Math.min(ans, dp[n][j]);
        System.out.println(ans);
    }
}
```

## **28. The Number of Imposters (Problem 2004C)**

### **Problem Description:**
n people, each says another is crewmate or impostor. Find max impostors possible.

### **Concept:**
- Graph coloring (bipartite)
- For each component, choose larger color

### **Java Code:**
```java
import java.util.*;

public class NumberOfImposters {
    static List<int[]>[] graph;
    static int[] color;
    static int[] cnt;
    static boolean possible;
    
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            int m = sc.nextInt();
            graph = new ArrayList[n + 1];
            for (int i = 1; i <= n; i++) graph[i] = new ArrayList<>();
            
            for (int i = 0; i < m; i++) {
                int u = sc.nextInt();
                int v = sc.nextInt();
                String s = sc.next();
                int w = s.equals("imposter") ? 1 : 0;
                graph[u].add(new int[]{v, w});
                graph[v].add(new int[]{u, w});
            }
            
            color = new int[n + 1];
            Arrays.fill(color, -1);
            possible = true;
            int ans = 0;
            
            for (int i = 1; i <= n; i++) {
                if (color[i] == -1) {
                    cnt = new int[2];
                    dfs(i, 0);
                    ans += Math.max(cnt[0], cnt[1]);
                }
            }
            
            System.out.println(possible ? ans : -1);
        }
    }
    
    private static void dfs(int node, int col) {
        color[node] = col;
        cnt[col]++;
        for (int[] edge : graph[node]) {
            int next = edge[0];
            int need = edge[1];
            if (color[next] == -1) {
                dfs(next, col ^ need);
            } else if ((color[next] ^ color[node]) != need) {
                possible = false;
            }
        }
    }
}
```

## **29. Moamen and XOR (Problem 2004D)**

Count arrays of length n with elements < 2^k such that AND >= XOR.

### **Concept:**

- Use combinatorics

```java
        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();
            
            long pow2 = 1;

            long ans = 0;
            if (n % 2 == 1) {
                ans = 1;
                for (int i = 0; i < k; i++) ans = ans * (pow2 + 1) % MOD;
            } else {
                long allSame = 1;
                for (int i = 0; i < k; i++) {
                    allSame = allSame * (pow2 - 1 + MOD) % MOD;
                }
                ans = allSame;
                long sum = 0;
                long mul = 1;
                for (int i = 0; i < k; i++) {
                    sum = (sum + mul) % MOD;
                    mul = mul * (pow2 - 1 + MOD) % MOD;
                }
                ans = (ans + sum) % MOD;
            }
            System.out.println(ans);
        }
    }
```
